static const char *va(const char *fmt, ...) {

static thread_local char tmp[0x10000];

static thread_local int at = 0;

char *ret = tmp+at;

va_list args;

va_start(args, fmt);

at += 1 + vsnprintf(ret, sizeof(tmp)-at-1, fmt, args);

va_end(args);

if(at > sizeof(tmp) - 0x400) {

at = 0;

}

return ret;

}
