float sea(vec2 uv, float chopiness) {

uv += noise(uv);

vec2 wv = 1.0 – abs(sin(uv));

vec2 swv = abs(cos(uv));

wv = mix(wv, swv, wv);

return pow(1.0 – pow(wv.x * wv.y,0.65), chopiness);
